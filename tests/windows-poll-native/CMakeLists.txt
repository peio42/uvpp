cmake_minimum_required(VERSION 3.14)
project(uvpp-windows-poll-native LANGUAGES C CXX)

if(NOT WIN32)
  message(FATAL_ERROR "This validation project requires native Windows")
endif()
if(NOT UVPP_NATIVE_LIBUV_SOURCE)
  message(FATAL_ERROR "Set UVPP_NATIVE_LIBUV_SOURCE to the audited libuv checkout")
endif()
set(UVPP_NATIVE_LIBUV_TARGET uv_a CACHE STRING "Native libuv target: uv_a or uv")
# Old audited libuv tags predate CMake 4's policy compatibility floor.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Legacy libuv policy compatibility")
set(LIBUV_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LIBUV_BUILD_BENCH OFF CACHE BOOL "" FORCE)
add_subdirectory("${UVPP_NATIVE_LIBUV_SOURCE}" libuv)
find_package(GTest CONFIG REQUIRED)
add_executable(uvpp-poll-native ../test-raw-poll-windows.cpp)
target_compile_features(uvpp-poll-native PRIVATE cxx_std_20)
target_include_directories(uvpp-poll-native PRIVATE ../../include)
target_link_libraries(uvpp-poll-native PRIVATE ${UVPP_NATIVE_LIBUV_TARGET} GTest::gtest_main)
target_compile_definitions(uvpp-poll-native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(uvpp-poll-native PRIVATE /W4 /WX /permissive- /Zc:__cplusplus /utf-8)
endif()
if(UVPP_NATIVE_LIBUV_TARGET STREQUAL "uv")
  add_custom_command(TARGET uvpp-poll-native POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      $<TARGET_FILE:uv> $<TARGET_FILE_DIR:uvpp-poll-native>)
endif()
enable_testing()
add_test(NAME windows-poll-native COMMAND uvpp-poll-native --gtest_brief=1 --gtest_output=xml:poll-native.xml)
set_tests_properties(windows-poll-native PROPERTIES TIMEOUT 60)
